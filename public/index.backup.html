PostgreSQL Relational Schema (Supabase / Neon DB)

Paste this SQL script into the Supabase SQL Editor or Neon Console to create tables, indexes, foreign key relationships, and automatic timestamp triggers. 

-- TechVerse Gear E-Commerce Schema for PostgreSQL (Supabase / Neon)

-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Users / Profiles Table (Linked with Google Auth)
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    google_id VARCHAR(255) UNIQUE,
    email VARCHAR(255) UNIQUE NOT NULL,
    full_name VARCHAR(255),
    avatar_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Products Table
CREATE TABLE IF NOT EXISTS public.products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(255) NOT NULL,
    category VARCHAR(100) NOT NULL,
    price NUMERIC(10, 2) NOT NULL,
    rating NUMERIC(2, 1) DEFAULT 5.0,
    description TEXT,
    image_url TEXT NOT NULL,
    stock_quantity INT DEFAULT 100,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Orders Master Table
CREATE TABLE IF NOT EXISTS public.orders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
    customer_email VARCHAR(255) NOT NULL,
    total_amount NUMERIC(10, 2) NOT NULL,
    status VARCHAR(50) DEFAULT 'Completed',
    shipping_address JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. Order Line Items Table
CREATE TABLE IF NOT EXISTS public.order_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    order_id UUID REFERENCES public.orders(id) ON DELETE CASCADE,
    product_id UUID REFERENCES public.products(id),
    product_title VARCHAR(255) NOT NULL,
    quantity INT NOT NULL,
    unit_price NUMERIC(10, 2) NOT NULL
);

-- Create Indexes for Query Performance
CREATE INDEX IF NOT EXISTS idx_products_category ON public.products(category);
CREATE INDEX IF NOT EXISTS idx_orders_customer_email ON public.orders(customer_email);
CREATE INDEX IF NOT EXISTS idx_order_items_order_id ON public.order_items(order_id);

Mailgun Node.js Email Handler

Node.js / Next.js Serverless Route Handler using official mailgun.js SDK to dispatch order receipts. 

// api/send-order-confirmation.js
import formData from 'form-data';
import Mailgun from 'mailgun.js';

const mailgun = new Mailgun(formData);
const mg = mailgun.client({
  username: 'api',
  key: process.env.MAILGUN_API_KEY || 'key-your-mailgun-api-key',
});

export default async function handler(req, res) {
  if (req.method !== 'POST') return res.status(405).send('Method Not Allowed');

  const { customerEmail, orderId, items, totalAmount, shippingAddress } = req.body;

  const itemRowsHtml = items.map(item => `
    <tr style="border-bottom: 1px solid #e2e8f0;">
      <td style="padding: 10px 0;"><b>${item.title}</b> (x${item.quantity})</td>
      <td style="padding: 10px 0; text-align: right;">$${(item.price * item.quantity).toFixed(2)}</td>
    </tr>
  `).join('');

  try {
    const mgResponse = await mg.messages.create(process.env.MAILGUN_DOMAIN, {
      from: "TechVerse Gear <orders@mg.techversegear.com>",
      to: [customerEmail],
      subject: `Order Confirmation #${orderId}`,
      html: `
        <div style="font-family: Arial, sans-serif; max-width: 600px; margin: auto; padding: 20px; border: 1px solid #e2e8f0; border-radius: 12px;">
          <h2 style="color: #0284c7; margin-top: 0;">Thank you for your order!</h2>
          <p>Your order <strong>#${orderId}</strong> has been received and is being processed.</p>
          <table style="width: 100%; border-collapse: collapse; font-size: 14px;">
            ${itemRowsHtml}
          </table>
          <h3 style="text-align: right; color: #0284c7; margin-top: 20px;">Total Amount Paid: $${parseFloat(totalAmount).toFixed(2)}</h3>
          <p style="font-size: 12px; color: #64748b;">Shipping to: ${shippingAddress}</p>
        </div>
      `
    });

    return res.status(200).json({ success: true, messageId: mgResponse.id });
  } catch (error) {
    return res.status(500).json({ error: error.message });
  }
}

Google Cloud Console OAuth 2.0 Checklist

Google Cloud Console Step-by-Step Setup:

  1. Log in to Google Cloud Console and create a new project.
  2. Go to APIs & Services > OAuth consent screen. Select External, enter your app name and developer support email.
  3. Add scopes: openid, email, profile.
  4. Go to Credentials > Create Credentials > OAuth Client ID. Choose Web Application.
  5. Set Authorized Redirect URIs for Supabase Auth: https://<your-project-ref>.supabase.co/auth/v1/callback.
// Frontend Integration with Supabase Client
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL,
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY
);

// Trigger Google SSO Login
export async function handleGoogleLogin() {
  const { data, error } = await supabase.auth.signInWithOAuth({
    provider: 'google',
    options: {
      redirectTo: window.location.origin
    }
  });
  if (error) console.error('OAuth Login Error:', error.message);
}
