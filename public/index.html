-- CURRENT SUPABASE DATABASE SCHEMA
-- =========================================================

-- PRODUCTS
CREATE TABLE IF NOT EXISTS public.products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  category TEXT NOT NULL,
  price NUMERIC NOT NULL,
  image_url TEXT,
  description TEXT,
  stock INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ORDERS
CREATE TABLE IF NOT EXISTS public.orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_email TEXT NOT NULL,
  total_amount NUMERIC NOT NULL,
  shipping_address JSONB NOT NULL,
  payment_method TEXT NOT NULL,
  status TEXT DEFAULT 'Processing',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ORDER ITEMS
CREATE TABLE IF NOT EXISTS public.order_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID REFERENCES public.orders(id) ON DELETE CASCADE,
  product_name TEXT NOT NULL,
  quantity INTEGER NOT NULL,
  price NUMERIC NOT NULL
);

-- PAYMENT INTENTS
CREATE TABLE IF NOT EXISTS public.payment_intents (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  reference TEXT UNIQUE NOT NULL,
  user_email TEXT NOT NULL,
  amount NUMERIC NOT NULL,
  currency TEXT DEFAULT 'NGN',
  shipping_address JSONB NOT NULL,
  items JSONB NOT NULL,
  status TEXT DEFAULT 'pending',
  order_id UUID REFERENCES public.orders(id),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- INDEXES
CREATE INDEX IF NOT EXISTS
  idx_payment_intents_reference
ON public.payment_intents(reference);

CREATE INDEX IF NOT EXISTS
  idx_orders_user_email
ON public.orders(user_email);

CREATE INDEX IF NOT EXISTS
  idx_order_items_order_id
ON public.order_items(order_id);

Payment → Order Architecture

Checkout Payment Intent Paystack Verify Order Created

Paystack Payment Flow

Active

TechVerse Gear uses Paystack to securely process customer payments. The backend calculates the authoritative order amount, initializes the transaction, and verifies the transaction before creating an order.

1

Initialize Payment

Customer checkout information is sent to the backend.

2

Server Calculates Total

Product prices and stock are retrieved from Supabase.

3

Paystack Checkout

Customer is redirected to Paystack to complete payment.

4

Server-Side Verification

Paystack confirms the transaction before the backend accepts it.

5

Order Finalized

The verified payment intent is finalized and the order is created.

Payment Security

Paystack secret credentials remain server-side. Product prices, payment amounts and payment status are validated by the backend.

Mailgun Email Integration

Integration Ready

Mailgun can be used by the backend for order confirmations, payment receipts and other transactional notifications. 

const formData = require("form-data");
const Mailgun = require("mailgun.js");

const mailgun = new Mailgun(formData);

const mg = mailgun.client({
  username: "api",
  key: process.env.MAILGUN_API_KEY
});

async function sendOrderConfirmation({
  customerEmail,
  orderId,
  totalAmount
}) {
  return await mg.messages.create(
    process.env.MAILGUN_DOMAIN,
    {
      from: "TechVerse Gear <orders@your-domain.com>",
      to: [customerEmail],
      subject: `Order Confirmation #${orderId}`,
      text: `
Thank you for your TechVerse Gear order.

Order ID: ${orderId}
Total Paid: ₦${Number(totalAmount).toLocaleString()}

Your order is being processed.
      `
    }
  );
}

Google Cloud Console Setup

  1. Open Google Cloud Console .
  2. Create or select the Google Cloud project.
  3. Configure the OAuth consent screen.
  4. Create an OAuth Client ID and choose Web Application.
  5. Configure the Supabase Auth callback: https://<your-project-ref>.supabase.co/auth/v1/callback
